 static void Gate()
 {
  string[] On(string user,string dir)=>new[]{"--stats-demo","--stats-expected-user="+user,"--stats-kit-dir="+dir};
  string code;
  Check(StatisticsKitGate.Evaluate(Array.Empty<string>(),"TwoSidesDemo",out code) is null&&code=="STATS_KIT_OFF","no arguments: transport stays OFF");
  Check(StatisticsKitGate.Evaluate(new[]{"--stats-expected-user=TwoSidesDemo","--stats-kit-dir=C:\\kit"},"TwoSidesDemo",out code) is null&&code=="STATS_KIT_OFF","expected user and folder WITHOUT the flag: OFF");
  Check(StatisticsKitGate.Evaluate(new[]{"--stats-demo"},"TwoSidesDemo",out code) is null&&code=="STATS_KIT_ARGUMENTS","flag without user/folder: rejected");
  Check(StatisticsKitGate.Evaluate(On("TwoSidesDemo","C:\\kit"),"OtherAccount",out code) is null&&code=="STATS_KIT_USER_MISMATCH","wrong Windows account: rejected");
  foreach(string forbidden in new[]{"luis2","TwoSidesTest","TwoSidesStatsTest","LUIS2"})
   Check(StatisticsKitGate.Evaluate(On(forbidden,"C:\\kit"),forbidden,out code) is null&&code=="STATS_KIT_USER_FORBIDDEN","accounts with real player data never activate: "+forbidden.ToLowerInvariant());
  Check(StatisticsKitGate.Evaluate(On("TwoSidesDemo","relative\\kit"),"TwoSidesDemo",out code) is null&&code=="STATS_KIT_DIRECTORY","relative folder: rejected");
  var ok=StatisticsKitGate.Evaluate(On("TwoSidesDemo","C:\\TwoSidesStatsKit\\hand-over"),"twosidesdemo",out code);
  Check(ok is not null&&code=="STATS_KIT_ENABLED"&&ok.KitDirectory=="C:\\TwoSidesStatsKit\\hand-over","expected account + flag + absolute folder: enabled (account compared case-insensitively)");
 }

 sealed class Server:IAsyncDisposable {
  readonly Process process;public readonly int Port;
  public Server(string storeDirectory,string kitDirectory){
   var start=new ProcessStartInfo("node"){RedirectStandardInput=true,RedirectStandardOutput=true,RedirectStandardError=true,UseShellExecute=false,CreateNoWindow=true};
   start.ArgumentList.Add("--import");start.ArgumentList.Add(new Uri(Path.GetFullPath("backend/tests/jwt/env-guard.js")).AbsoluteUri);start.ArgumentList.Add(Path.GetFullPath("backend/tests/b87/kit-e2e-server.js"));start.ArgumentList.Add(storeDirectory);start.ArgumentList.Add(kitDirectory);
   process=Process.Start(start)!;
   string? line=process.StandardOutput.ReadLineAsync().WaitAsync(TimeSpan.FromSeconds(10)).GetAwaiter().GetResult();
   if(line is null)throw new Exception("Express kit server bootstrap failed");using var json=JsonDocument.Parse(line);Port=json.RootElement.GetProperty("port").GetInt32();
  }
  public async Task<long> Total(){await process.StandardInput.WriteLineAsync("summary");using var json=JsonDocument.Parse((await process.StandardOutput.ReadLineAsync().WaitAsync(TimeSpan.FromSeconds(5)))!);return json.RootElement.TryGetProperty("all|events",out var value)?value.GetInt64():0;}

 }
 static async Task<StatisticsKit?> WaitForKit(string dir,string id){for(int i=0;i<100;i++){var kit=StatisticsKitFiles.TryRead(dir,Now,id);if(kit is not null)return kit;await Task.Delay(100);}return null;}

 static async Task EndToEnd()
 {
  await using var f=new Fixture();await f.Start();
  string installation;using(var c=f.Db.Open()){using var q=c.CreateCommand();q.CommandText="SELECT installation_id FROM statistics_installation;";installation=(string)q.ExecuteScalar()!;}
  var outbox=f.Adapter.StatisticalOutbox;Check(outbox is not null&&outbox.GetInstallationId()==installation,"accessor GetInstallationId returns the stored installation id");
  string store=TempDir("twosides-b79-"),kit=TempDir("twosides-b87-hand-over-");
  try{
   await using(var server=new Server(store,kit))
   {
    // No kit yet: the transport pauses by itself and sends nothing.
    await using(var early=new StatisticsTransport(f.Adapter,new KitStatisticsCredentials(kit,installation),new Uri("http://127.0.0.1:"+server.Port+"/"),syntheticLoopback:true))
    {
     string tick=await early.TickAsync();Check(tick=="STATISTICS_AUTH_REQUIRED"&&await server.Total()==0,"before the hand-over: no credential, nothing sent");
    }
    Check(!File.Exists(Path.Combine(kit,"kit.json")),"the server issues no token before the game announces itself");
    StatisticsKitFiles.WriteInstallation(kit,installation);
    var published=await WaitForKit(kit,installation);Check(published is not null&&published.Audience=="http://127.0.0.1:"+server.Port,"the real server publishes a loopback kit for exactly this installation");
    Check(f.N("SELECT count(*) FROM statistics_outbox WHERE state<>'confirmed';")==5,"five pending statistical events before delivery");
    await using var transport=new StatisticsTransport(f.Adapter,new KitStatisticsCredentials(kit,installation),new Uri(published!.Audience+"/"),syntheticLoopback:true);
    string result=await transport.TickAsync();
    Check(result=="STATISTICS_RECEIPTS_APPLIED","real hand-over: receipts applied ("+result+")");
    Check(f.N("SELECT count(*) FROM statistics_outbox WHERE state='confirmed';")==5&&await server.Total()==5,"all five events confirmed locally and counted once by Express");
    Check((await f.Adapter.LoadGameAsync(f.Owner,f.Game)).Data!.Biomes.Single(b=>b.BiomeId=="ocean").Status=="completed","game progress untouched by delivery");
    // The kit is removed when the server closes.
   }
   Check(!File.Exists(Path.Combine(kit,"kit.json")),"kit.json is deleted when the server stops");
  }finally{Remove(kit);foreach(var file in Directory.GetFiles(store,"*",SearchOption.AllDirectories))File.Delete(file);foreach(var sub in Directory.GetDirectories(store,"*",SearchOption.AllDirectories).OrderByDescending(x=>x.Length))Directory.Delete(sub);Directory.Delete(store);}
 }
