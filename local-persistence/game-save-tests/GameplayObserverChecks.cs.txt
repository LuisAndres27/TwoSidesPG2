    internal sealed class Store:IGameplayTelemetryStore
    {
        internal readonly GameplayEventService Service;
        internal bool Fail,LostResponse,Hang;
        internal readonly List<GameplayEvent> Attempts=new();
        internal readonly List<GameplayInterval> IntervalAttempts=new();
        internal TaskCompletionSource<bool>? Gate;
        internal Store(ISaveDatabase db)=>Service=new(db);
        public Task<LocalSaveResult<IReadOnlyList<GameplaySession>>> ReadSessionsAsync(string owner)=>Task.FromResult(new LocalSaveResult<IReadOnlyList<GameplaySession>>(true,"SESSIONS",Service.GetSessions(owner)));
        public async Task<LocalSaveResult<GameplayWriteReceipt>> WriteEventAsync(GameplayEvent value)
        {
            lock(Attempts)Attempts.Add(value);
            if(Hang){Gate??=new();await Gate.Task;}
            if(Fail)return new(false,"IO_ERROR");
            try{var r=Service.RecordEvent(value);if(LostResponse){LostResponse=false;return new(false,"IO_ERROR");}return new(true,"EVENT",r);}catch(SaveException e){return new(false,e.Code);}
        }
        public Task<LocalSaveResult<GameplayWriteReceipt>> WriteIntervalAsync(GameplayInterval value)
        {
            lock(IntervalAttempts)IntervalAttempts.Add(value);
            if(Fail)return Task.FromResult(new LocalSaveResult<GameplayWriteReceipt>(false,"IO_ERROR"));
            try{return Task.FromResult(new LocalSaveResult<GameplayWriteReceipt>(true,"INTERVAL",Service.RecordInterval(value)));}catch(SaveException e){return Task.FromResult(new LocalSaveResult<GameplayWriteReceipt>(false,e.Code));}
        }
        public Task<LocalSaveResult<GameplayWriteReceipt>> InterruptAsync(string owner,string session,string eventId,DateTimeOffset discovered)
        {try{return Task.FromResult(new LocalSaveResult<GameplayWriteReceipt>(true,"INTERRUPTED",Service.InterruptSession(owner,session,eventId,discovered)));}catch(SaveException e){return Task.FromResult(new LocalSaveResult<GameplayWriteReceipt>(false,e.Code));}}
    }
