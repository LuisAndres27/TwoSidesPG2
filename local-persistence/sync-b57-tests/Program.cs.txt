    static async Task Modes()

        Check(SyncTestStorageMode.FromArguments(new[]{SyncLocalMode.Flag,SyncTestStorageMode.Flag})==SyncTestStorageRequest.Sandbox,"legacy HTTP sandbox unchanged");

    static async Task Sandbox()
    {
        string path;
        using(var storage=GameTestStorage.FromArguments(Https)!)
        {
            path=storage.Database.Path;
            string prefix=Path.Combine(SyntheticSandbox.AuthorizedProjectRoot,".work","databases")+Path.DirectorySeparatorChar;
            Check(path.StartsWith(prefix,StringComparison.OrdinalIgnoreCase)&&Path.GetExtension(path)==".sqlite3","only a newly owned disposable database");
            Check((File.GetAttributes(path)&FileAttributes.ReparsePoint)==0,"synthetic database is a normal file");
            await using var adapter=storage.CreateAdapter();
            // Only rejected strings; no inspection, hashes or opening of real files.
            foreach(string rejectedPath in new[]{GamePersistenceAdapter.RuntimeStoragePath,GamePersistenceAdapter.StoragePath,path,"",@"C:\Users\luis2\AppData\Roaming\Godot\app_userdata\TwoSides\save.sqlite3"})
                Check(!(await adapter.OpenAsync(rejectedPath)).Success,"adapter rejects every path except its synthetic capability token");
            Check((await adapter.OpenAsync(GameTestStorage.OpenPath)).Success&&adapter.SyncStorage is not null,"synthetic capability opens only the created database");
            await adapter.DrainOutboxAsync(TimeSpan.FromSeconds(1));
            Check(adapter.SyncStorage is null,"closed synthetic adapter cannot keep syncing");
        }
        Check(!File.Exists(path)&&!Directory.Exists(Path.GetDirectoryName(path)!),"only this sandbox removed at disposal");
        using var a=GameTestStorage.FromArguments(Https)!;using var b=GameTestStorage.FromArguments(Https)!;
        Check(a.Database.Path!=b.Database.Path,"each test run gets fresh independent storage");
    }
    static async Task RejectInvalidCertificate()
    {
        // TLS server double, in RAM on an ephemeral loopback port. No Caddy/CA installation or trust bypass.
        using var key=RSA.Create(2048);
        var req=new CertificateRequest("CN=localhost",key,HashAlgorithmName.SHA256,RSASignaturePadding.Pkcs1);
        var san=new SubjectAlternativeNameBuilder();san.AddDnsName("localhost");req.CertificateExtensions.Add(san.Build());
        req.CertificateExtensions.Add(new X509BasicConstraintsExtension(false,false,0,true));
        using var cert=req.CreateSelfSigned(DateTimeOffset.UtcNow.AddMinutes(-1),DateTimeOffset.UtcNow.AddMinutes(5));
        using var chain=new X509Chain();chain.ChainPolicy.RevocationMode=X509RevocationMode.NoCheck;
        Check(!chain.Build(cert),"self-signed certificate is not trusted by OS");
        var listener=new TcpListener(IPAddress.Loopback,0);listener.Start();
        using var limit=new CancellationTokenSource(TimeSpan.FromSeconds(10));
        try
        {
            int port=((IPEndPoint)listener.LocalEndpoint).Port;
            var server=Task.Run(async()=>{
                using var tcp=await listener.AcceptTcpClientAsync(limit.Token);
                using var tls=new SslStream(tcp.GetStream(),false);
                try{await tls.AuthenticateAsServerAsync(new SslServerAuthenticationOptions{ServerCertificate=cert,EnabledSslProtocols=SslProtocols.Tls12|SslProtocols.Tls13},limit.Token);}
                catch(AuthenticationException){}catch(IOException){}
            });
            using var transport=new SyncHttpTransport(new SyncHttpOptions(new Uri($"https://localhost:{port}/"),TimeSpan.FromSeconds(5)));
            var response=await transport.SendAsync(HttpMethod.Get,"/api/health",null,null,limit.Token);
            Check(response.Status==0&&response.Failure==SyncHttpFailure.NetworkUnavailable,"production HttpClient rejects invalid certificate without TLS callback");
            await server;
        }
        finally{listener.Stop();}
    }

        }
        catch(Exception error){Console.Error.WriteLine("B57_RESULT=FAIL "+error.GetType().Name+": "+error.Message);return 1;}
    }
