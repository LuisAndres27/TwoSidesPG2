    private static void Error(string code,Action action)
    {try{action();throw new Exception("Expected "+code);}catch(SaveException e){if(e.Code!=code)throw new Exception($"Expected {code}, got {e.Code}");}}
    private sealed class F:IDisposable
    {
        internal readonly SyntheticSandbox Sandbox=new(SyntheticSandbox.AuthorizedProjectRoot);
        internal readonly SyntheticDatabase Db;
        internal readonly GameSaveService Save;
        internal readonly string Owner=Id(),Game=Id();
        internal string Session="",Run="",Biome="ocean";
        internal GameplayEventService Events=null!;
        internal F(bool migrate=true)
        {
            Db=Sandbox.CreateDatabase();SchemaManager.Initialize(Db);Save=new(Db,RealGameCatalog.InitialState());
            Save.CreateLocalProfile(Id(),Owner,"Eventos sintéticos");Save.CreateNewGame(new(Id(),Owner,Game));
            Save.EnterBiome(new(Id(),Owner,Game,0,"ocean","human"));
            Save.RecordDecision(new(Id(),Id(),Owner,Game,1,"ocean","ocean_waste_disposal","reject","healthy","{\"stateKind\":\"healthy\"}"));
            if(migrate){GameplayEventSchema.MigrateSynthetic(Db);Events=new(Db);}
        }
        internal GameplayEvent SessionEvent(GameplayEventKind kind,long offset,string? eventId=null,GameplaySessionEnd? end=null)
            =>new(eventId??Id(),Owner,Session,kind,DateTimeOffset.UnixEpoch.AddMilliseconds(offset),offset,SessionEnd:end);
        internal GameplayEvent AnimalEvent(GameplayEventKind kind,long offset,string? id=null)
            =>new(id??Id(),Owner,Session,kind,DateTimeOffset.UnixEpoch.AddMilliseconds(offset),offset,Game,Biome,Run);
        internal GameplayEvent Start()
        {Session=Id();var e=SessionEvent(GameplayEventKind.SESSION_STARTED,0);Events.RecordEvent(e);return e;}
        internal GameplayEvent Animal(long offset=0)
        {if(Session=="")Start();Run=Id();var e=AnimalEvent(GameplayEventKind.ANIMAL_SEGMENT_STARTED,offset,Run);Events.RecordEvent(e);return e;}
        internal GameplayInterval Interval(long start,long end,GameplayTimeState state=GameplayTimeState.ActiveAnimal)
            =>new(Id(),Owner,Session,start,end,state,Game,Biome,state==GameplayTimeState.ActiveHuman?null:Run);
        internal string Hash()=>Convert.ToHexString(SHA256.HashData(File.ReadAllBytes(Db.Path)));
        internal long Number(string sql){using var c=Db.Open();using var q=c.CreateCommand();q.CommandText=sql;return Convert.ToInt64(q.ExecuteScalar());}
        internal void Sql(string sql){using var c=Db.Open();using var q=c.CreateCommand();q.CommandText=sql;q.ExecuteNonQuery();}
        public void Dispose()=>Sandbox.Dispose();
    }
    private static async Task Case(string name,Func<F,Task> work,bool migrate=true)
    {using var f=new F(migrate);await work(f);_count++;Console.WriteLine("PASS "+name);}

    {var service=new GameplayEventService(new RuntimeMenuChecks.ReadOnlyFixture(path));return service.GetEvents(owner,game).Count==2 && service.GetTotals(owner,game).Deaths.Value==1?0:71;}
