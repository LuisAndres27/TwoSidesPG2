using System;
using System.Collections.Generic;
using System.IO;
using System.Runtime.InteropServices;
using System.Security.Cryptography;
using System.Text;
using System.Text.Json;

using TwoSides.LocalPersistence;

namespace TwoSides.Integration.SyncHttp;

// Windows-only persistent store for the sync credential of ONE (local player, installation, generation) binding, protected with DPAPI in the
// CurrentUser scope (CryptProtectData; no LOCAL_MACHINE flag, no UI). There is no key in the game: Windows derives it from the signed-in account.
//
//  * One file per binding: <dir>/<first 32 hex of SHA-256(binding)>.cred. The file holds ONLY the DPAPI blob; inside, a small JSON document
//    {v, player, installation, generation, registrationId, account, credential}. Nothing readable is on disk, not even the player id.
//  * The binding string is also passed to DPAPI as optional entropy and re-checked inside the decrypted document, so a blob copied to another
//    binding's file name, or edited, is refused instead of being used for a different player.
//  * FAIL CLOSED: if a file exists but cannot be used (wrong user/machine, corrupt, tampered, too big, a link, wrong binding), the binding is
//    "unavailable". The store never silently overwrites it and the session manager never registers a fresh account in its place: that would
//    create a second identity for the same player. Synchronization pauses (operations stay retryable); the game itself is unaffected.
//    Only an explicit Forget() clears the situation.
//  * Forget() overwrites the file with random bytes before deleting it (best effort on SSDs / cloud-synced folders).
//  * Writes are atomic (temp file + move). Reads/writes are synchronous: call them from the sync worker, never the Godot frame thread.
public sealed class DpapiSyncSecretStore:ISyncSecretStore
{
    private const int MaxFileBytes=8192;
    private sealed record Entry(string Credential,string? Account,string? RegistrationId);
    private readonly string _directory;
    private readonly object _lock=new();
    private readonly Dictionary<string,Entry> _cache=new();
    private readonly HashSet<string> _unusable=new();

    public DpapiSyncSecretStore(string directory)
    {
        SaveValidation.Require(OperatingSystem.IsWindows(),"UNSUPPORTED_PLATFORM");
        SaveValidation.Require(Path.IsPathFullyQualified(directory),"UNSAFE_PATH");
        _directory=Path.GetFullPath(directory);
        try
        {
            if(Directory.Exists(_directory)){SaveValidation.Require((File.GetAttributes(_directory)&FileAttributes.ReparsePoint)==0,"UNSAFE_PATH");}
            else Directory.CreateDirectory(_directory);
        }
        catch(Exception error) when(error is IOException or UnauthorizedAccessException){throw new SaveException("CREDENTIAL_STORE_UNAVAILABLE");}
    }

    private static string Key(SyncBinding b)=>b.LocalPlayerId+"|"+b.InstallationId+"|"+b.BindingGeneration;
    private string PathFor(SyncBinding b)=>Path.Combine(_directory,Convert.ToHexString(SHA256.HashData(Encoding.UTF8.GetBytes(Key(b))))[..32].ToLowerInvariant()+".cred");
    private static byte[] Entropy(SyncBinding b)=>Encoding.UTF8.GetBytes("TwoSides.SyncCredential.v1\n"+Key(b));

    public bool IsUnavailable(SyncBinding binding){lock(_lock){Load(binding);return _unusable.Contains(Key(binding));}}
    public string? ReadCredential(SyncBinding binding){lock(_lock){return Load(binding)?.Credential;}}
    public string? ReadAccount(SyncBinding binding){lock(_lock){return Load(binding)?.Account;}}
    public string? ReadRegistrationId(SyncBinding binding){lock(_lock){return Load(binding)?.RegistrationId;}}

    public void WriteCredential(SyncBinding binding,string credential)
    {
        lock(_lock)
        {
            var current=Load(binding);RequireUsable(binding);
            if(current is not null&&current.Credential==credential)return;
            Save(binding,new(credential,null,current?.RegistrationId));                  // a new credential means the old account no longer applies
        }
    }
    public void WriteAccount(SyncBinding binding,string account)
    {
        lock(_lock){var current=Load(binding);RequireUsable(binding);SaveValidation.Require(current is not null,"CREDENTIAL_UNAVAILABLE");
            if(current!.Account!=account)Save(binding,current with{Account=account});}
    }
    public void WriteRegistrationId(SyncBinding binding,string registrationId)
    {
        lock(_lock){var current=Load(binding);RequireUsable(binding);SaveValidation.Require(current is not null,"CREDENTIAL_UNAVAILABLE");
            if(current!.RegistrationId!=registrationId)Save(binding,current with{RegistrationId=registrationId});}
    }

    // Explicit local removal (sign out of this device). Also the only way out of an unusable state.
    public void Forget(SyncBinding binding)
    {
        lock(_lock)
        {
            string path=PathFor(binding);
            _cache.Remove(Key(binding));_unusable.Remove(Key(binding));
            try
            {
                if(!File.Exists(path))return;
                var attributes=File.GetAttributes(path);
                if((attributes&FileAttributes.ReparsePoint)!=0){File.Delete(path);return;}          // never write through a link
                long length=Math.Clamp(new FileInfo(path).Length,64,MaxFileBytes);
                using(var stream=new FileStream(path,FileMode.Open,FileAccess.Write,FileShare.None))
                {stream.Write(RandomNumberGenerator.GetBytes((int)length));stream.Flush(true);}
                File.Delete(path);
            }
            catch(Exception error) when(error is IOException or UnauthorizedAccessException){throw new SaveException("CREDENTIAL_STORE_UNAVAILABLE");}
        }
    }

    private void RequireUsable(SyncBinding binding){if(_unusable.Contains(Key(binding)))throw new SaveException("CREDENTIAL_UNAVAILABLE");}

    private Entry? Load(SyncBinding binding)
    {
        string key=Key(binding);
        if(_cache.TryGetValue(key,out var cached))return cached;
        if(_unusable.Contains(key))return null;
        string path=PathFor(binding);
        try
        {
            if(!File.Exists(path))return null;
            var info=new FileInfo(path);
            if((info.Attributes&FileAttributes.ReparsePoint)!=0||info.Length is <16 or >MaxFileBytes)return Unusable(key);
            byte[] blob=File.ReadAllBytes(path);
            byte[]? plain=Dpapi.Unprotect(blob,Entropy(binding));
            if(plain is null)return Unusable(key);
            try
            {
                using var doc=JsonDocument.Parse(plain);var root=doc.RootElement;
                if(root.ValueKind!=JsonValueKind.Object||Text(root,"player")!=binding.LocalPlayerId||Text(root,"installation")!=binding.InstallationId||
                   !root.TryGetProperty("generation",out var generation)||!generation.TryGetInt64(out long g)||g!=binding.BindingGeneration||
                   !root.TryGetProperty("v",out var v)||!v.TryGetInt32(out int version)||version!=1)return Unusable(key);
                string? credential=Text(root,"credential");
                if(string.IsNullOrEmpty(credential))return Unusable(key);
                var entry=new Entry(credential,Text(root,"account"),Text(root,"registrationId"));
                _cache[key]=entry;return entry;
            }
            catch(JsonException){return Unusable(key);}
            finally{Array.Clear(plain);}
        }
        catch(Exception error) when(error is IOException or UnauthorizedAccessException){return Unusable(key);}
    }

    private Entry? Unusable(string key){_unusable.Add(key);return null;}
    private static string? Text(JsonElement o,string name)=>o.TryGetProperty(name,out var v)&&v.ValueKind==JsonValueKind.String?v.GetString():null;

    private void Save(SyncBinding binding,Entry entry)
    {
        byte[] plain=JsonSerializer.SerializeToUtf8Bytes(new{v=1,player=binding.LocalPlayerId,installation=binding.InstallationId,generation=binding.BindingGeneration,
            registrationId=entry.RegistrationId,account=entry.Account,credential=entry.Credential});
        try
        {
            byte[] blob=Dpapi.Protect(plain,Entropy(binding))??throw new SaveException("CREDENTIAL_STORE_UNAVAILABLE");
            string path=PathFor(binding),temp=path+".tmp";
            try
            {
                if(File.Exists(path))SaveValidation.Require((File.GetAttributes(path)&FileAttributes.ReparsePoint)==0,"UNSAFE_PATH");
                if(File.Exists(temp))File.Delete(temp);
                using(var stream=new FileStream(temp,FileMode.CreateNew,FileAccess.Write,FileShare.None)){stream.Write(blob);stream.Flush(true);}
                File.Move(temp,path,true);
            }
            catch(Exception error) when(error is IOException or UnauthorizedAccessException){throw new SaveException("CREDENTIAL_STORE_UNAVAILABLE");}
            _cache[Key(binding)]=entry;
        }
        finally{Array.Clear(plain);}
    }

    private static class Dpapi
    {
        private const int UiForbidden=0x1;
        [StructLayout(LayoutKind.Sequential)] private struct Blob{public int Size;public IntPtr Data;}
        [DllImport("crypt32.dll",SetLastError=true,CharSet=CharSet.Unicode)]
        private static extern bool CryptProtectData(ref Blob input,string? description,ref Blob entropy,IntPtr reserved,IntPtr prompt,int flags,out Blob output);
        [DllImport("crypt32.dll",SetLastError=true,CharSet=CharSet.Unicode)]
        private static extern bool CryptUnprotectData(ref Blob input,IntPtr description,ref Blob entropy,IntPtr reserved,IntPtr prompt,int flags,out Blob output);
        [DllImport("kernel32.dll")] private static extern IntPtr LocalFree(IntPtr handle);

        public static byte[]? Protect(byte[] data,byte[] entropy)=>Run(data,entropy,true);
        public static byte[]? Unprotect(byte[] data,byte[] entropy)=>Run(data,entropy,false);

        private static byte[]? Run(byte[] data,byte[] entropy,bool protect)
        {
            IntPtr input=IntPtr.Zero,extra=IntPtr.Zero;
            try
            {
                input=Marshal.AllocHGlobal(Math.Max(1,data.Length));Marshal.Copy(data,0,input,data.Length);
                extra=Marshal.AllocHGlobal(Math.Max(1,entropy.Length));Marshal.Copy(entropy,0,extra,entropy.Length);
                var source=new Blob{Size=data.Length,Data=input};var salt=new Blob{Size=entropy.Length,Data=extra};
                Blob result;
                bool ok=protect?CryptProtectData(ref source,null,ref salt,IntPtr.Zero,IntPtr.Zero,UiForbidden,out result)
                    :CryptUnprotectData(ref source,IntPtr.Zero,ref salt,IntPtr.Zero,IntPtr.Zero,UiForbidden,out result);
                if(!ok)return null;
                try{var bytes=new byte[result.Size];Marshal.Copy(result.Data,bytes,0,result.Size);return bytes;}
                finally{if(result.Data!=IntPtr.Zero){Marshal.Copy(new byte[result.Size],0,result.Data,result.Size);LocalFree(result.Data);}}
            }
            finally
            {
                if(input!=IntPtr.Zero){Marshal.Copy(new byte[data.Length],0,input,data.Length);Marshal.FreeHGlobal(input);}
                if(extra!=IntPtr.Zero)Marshal.FreeHGlobal(extra);
            }
        }
    }
}
