using System;
using System.Threading.Tasks;

internal static class Program
{
    public static async Task<int> Main(string[] args)
    {
        try
        {
            if(args.Length>0 && args[0] is "--make-synthetic-store" or "--extend-synthetic-store")return SyntheticStoreMaker.Run(args);
            return await ProductionStorageChecks.Run(args);
        }
        catch(Exception error){Console.WriteLine("FAIL "+error.GetType().Name+": "+error.Message);return 99;}
    }
}
