    // ---- the real Express offline backend as a child process ----
    private sealed class OfflineServer:IDisposable
    {
        private Process? _process;
        internal int Port;
        internal string Url=>$"http://127.0.0.1:{Port}";
        internal string ErrorOutput="";
        private static int FreePort(){var l=new TcpListener(IPAddress.Loopback,0);l.Start();int p=((IPEndPoint)l.LocalEndpoint).Port;l.Stop();return p;}
        internal static string BackendRoot=>Path.GetFullPath(Path.Combine(SyntheticSandbox.AuthorizedProjectRoot,"..","backend"));
        internal void Start(int? port=null)
        {
            Port=port??FreePort();
            var info=new ProcessStartInfo(@"C:\Program Files\nodejs\node.exe"){WorkingDirectory=BackendRoot,UseShellExecute=false,CreateNoWindow=true,
                RedirectStandardInput=true,RedirectStandardOutput=true,RedirectStandardError=true};
            foreach(string a in new[]{"scripts/run-offline-backend.js","--port",Port.ToString(CultureInfo.InvariantCulture),"--exit-on-stdin-close"})info.ArgumentList.Add(a);
            info.Environment.Clear();string win=Environment.GetFolderPath(Environment.SpecialFolder.Windows);
            info.Environment["SystemRoot"]=win;info.Environment["PATH"]=Path.Combine(win,"System32");info.Environment["NODE_ENV"]="test";
            _process=Process.Start(info)!;
            var line=_process.StandardOutput.ReadLineAsync();
            if(!line.Wait(30000)||line.Result is null||!line.Result.Contains("\"offlineBackend\":\"listening\"")||!line.Result.Contains(Url))throw new Exception("Offline backend did not start");
            _=_process.StandardOutput.ReadToEndAsync();
            var err=_process.StandardError.ReadToEndAsync();_ = err.ContinueWith(t=>ErrorOutput=t.Result);
        }
        // Orderly stop through the launcher's stdin command; returns the exit code (0 = clean).
        internal int Stop()
        {
            if(_process is null)return -1;
            try{_process.StandardInput.WriteLine("shutdown");_process.StandardInput.Flush();}catch(IOException){}
            if(!_process.WaitForExit(20000)){_process.Kill(true);_process.WaitForExit();return -2;}
            int code=_process.ExitCode;_process.Dispose();_process=null;return code;
        }
        // Abrupt interruption (crash / lost connection).
        internal void Kill(){if(_process is null)return;try{_process.Kill(true);_process.WaitForExit();}catch(InvalidOperationException){}_process.Dispose();_process=null;}
        public void Dispose(){if(_process is not null)Kill();}
    }

    // Accepts TCP connections and never answers: the request can only end by timeout.
    private sealed class HangingServer:IDisposable
    {
        private readonly TcpListener _listener=new(IPAddress.Loopback,0);
        private readonly List<TcpClient> _clients=new();
        internal int Port=>((IPEndPoint)_listener.LocalEndpoint).Port;
        internal HangingServer(){_listener.Start();_=Task.Run(async()=>{try{while(true){var client=await _listener.AcceptTcpClientAsync();lock(_clients)_clients.Add(client);}}catch(Exception){}});}
        public void Dispose(){_listener.Stop();lock(_clients)foreach(var c in _clients)c.Dispose();}
    }

    // Wraps the real transport handler: counts calls, can lose a response after the server committed, and can alter bytes in flight.
    private sealed class TapHandler:DelegatingHandler
    {
        internal int Batches,Queries,Registrations,Logins,LastBatchOperations;
        internal readonly List<string> BatchTokens=new();
        internal bool DropNextBatchResponse;
        internal Func<string,string>? RewriteRequest,RewriteResponse;
        protected override async Task<HttpResponseMessage> SendAsync(HttpRequestMessage request,CancellationToken cancellation)
        {
            string path=request.RequestUri!.AbsolutePath;bool batch=path.EndsWith("/sync/batches",StringComparison.Ordinal);
            if(path.EndsWith("/receipts/query",StringComparison.Ordinal))Interlocked.Increment(ref Queries);
            else if(path.EndsWith("/technical/registrations",StringComparison.Ordinal))Interlocked.Increment(ref Registrations);
            else if(path.EndsWith("/technical/sessions",StringComparison.Ordinal))Interlocked.Increment(ref Logins);
            if(batch)
            {
                Interlocked.Increment(ref Batches);
                lock(BatchTokens)BatchTokens.Add(request.Headers.Authorization?.Parameter??"");
                string text=await request.Content!.ReadAsStringAsync(cancellation);
                using(var doc=JsonDocument.Parse(text))LastBatchOperations=doc.RootElement.GetProperty("operations").GetArrayLength();
                if(RewriteRequest is not null)
                {
                    request.Content=new StringContent(RewriteRequest(text),new UTF8Encoding(false),"application/json");
                }
            }
            var response=await base.SendAsync(request,cancellation);
            if(batch && DropNextBatchResponse){DropNextBatchResponse=false;response.Dispose();throw new HttpRequestException("response lost after commit");}
            if(batch && RewriteResponse is not null)
            {
                string text=await response.Content.ReadAsStringAsync(cancellation);
                return new HttpResponseMessage(response.StatusCode){Content=new StringContent(RewriteResponse(text),new UTF8Encoding(false),"application/json")};
            }
            return response;
        }
    }

    private static async Task Case(string label,Func<F,Task> body){using var f=new F();await body(f);_count++;Console.WriteLine("PASS "+label);}
    private static async Task<SyncPassSummary> Pass(Client c)=>await c.Dispatcher.RunPassAsync(CancellationToken.None);
    private static async Task<List<SyncPassSummary>> Drain(Client c,int max=6)
    {var all=new List<SyncPassSummary>();for(int i=0;i<max;i++){var s=await Pass(c);all.Add(s);if(s.Code=="SYNC_IDLE")break;}return all;}
