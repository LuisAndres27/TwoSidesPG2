 static async Task<HttpResponseMessage> Reply(HttpRequestMessage req,Func<List<Dictionary<string,string>>,List<Dictionary<string,string>>>? transform=null){
  string text=await req.Content!.ReadAsStringAsync();using var body=JsonDocument.Parse(text);var receipts=new List<Dictionary<string,string>>();
  Check(body.RootElement.GetProperty("Contract").GetString()==StatisticsTransport.Contract,"wire contract explicit");
  Check(body.RootElement.GetProperty("Events").GetArrayLength()<=5,"wire batch bounded");
  foreach(var item in body.RootElement.GetProperty("Events").EnumerateArray()){
   using var e=JsonDocument.Parse(item.GetProperty("EventJson").GetString()!);
   receipts.Add(new(){{"EventId",e.RootElement.GetProperty("EventId").GetString()!},{"PayloadHash",item.GetProperty("PayloadHash").GetString()!},{"Status","accepted"},{"Code","ok"}});
  }
  Check(!text.Contains("PRIVATE-SYNTHETIC")&&!text.Contains("display_name")&&!text.Contains("game.report"),"wire has no profile/save payload");
  return new(HttpStatusCode.OK){Content=new StringContent(JsonSerializer.Serialize(new{Contract=StatisticsTransport.Contract,Receipts=transform?.Invoke(receipts)??receipts}),Encoding.UTF8,"application/json")};
 }
 static async Task Receipts(){

   await using var f=new Fixture();await f.Start();var h=new Handler();h.Respond=async(r,c)=>mode=="malformed"?new(HttpStatusCode.OK){Content=new StringContent("{}")}:await Reply(r,items=>{
    if(mode=="reverse")items.Reverse();if(mode=="partial")return items.Take(1).ToList();if(mode=="empty")return new();
    if(mode=="wrong-id")items[0]["EventId"]=Id();if(mode=="wrong-hash")items[0]["PayloadHash"]=new string('0',64);if(mode=="duplicate")items[1]=new(items[0]);
    if(mode=="permanent"){items[0]["Status"]="rejected";items[0]["Code"]="event_expired";}return items;
   });await using var w=f.Worker(h);await w.TickAsync();

   Check(f.N("SELECT count(*) FROM statistics_outbox WHERE state='confirmed';")==expected,"individual receipts: "+mode);
   Check(f.N("SELECT count(*) FROM statistics_events;")==5,"sending never creates events: "+mode);

  }

 static async Task Errors(){
  foreach(int status in new[]{401,403,400,404,429,500,503,302}){
   await using var f=new Fixture();await f.Start();var h=new Handler{Respond=(r,c)=>{var response=new HttpResponseMessage((HttpStatusCode)status);if(status==429)response.Headers.RetryAfter=new RetryConditionHeaderValue(TimeSpan.FromSeconds(240));return Task.FromResult(response);}};
   await using var w=f.Worker(h);await w.TickAsync();Check(f.N("SELECT count(*) FROM statistics_outbox WHERE state='confirmed';")==0,"HTTP "+status+" never acknowledges");
   Check(f.N("SELECT count(*) FROM statistics_outbox WHERE state='retryable_failure';")==5,"HTTP failure retains five pending");
   if(status==429)Check(f.N("SELECT min(next_attempt_ms) FROM statistics_outbox;")>=f.Now+240000,"Retry-After respected");
   if(status==401){await w.TickAsync();Check(h.Calls==1,"auth failure pauses without loop");await w.ResumeAuthenticationAsync();h.Respond=(r,c)=>Reply(r);await w.TickAsync();Check(f.N("SELECT count(*) FROM statistics_outbox WHERE state='confirmed';")==5,"explicit auth renewal resumes same IDs");}
  }

  }
