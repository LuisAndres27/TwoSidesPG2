internal static class LocalStatisticsChecks
{
    private static int _count;
    private static string Id()=>Guid.NewGuid().ToString("D");
    private static readonly string[] Biomes={"ocean","jungle","desert","forest","tundra"};
    private static readonly string[] Keys={"ocean_waste_disposal","jungle_transport_route","desert_cat_abandonment","forest_industrial_pesticides","tundra_fossil_extraction"};
    private static void Check(bool ok){if(!ok)throw new Exception("Local statistics assertion failed");}
    private static void Error(string code,Action action)
    {try{action();throw new Exception("Expected "+code);}catch(SaveException e){Check(e.Code==code);}}
    private sealed class ReadOnly:ISaveDatabase
    {
        private readonly ISaveDatabase _db;internal int Opens;
        internal ReadOnly(ISaveDatabase db)=>_db=db;
        public SqliteConnection Open()
        {
            Opens++;var c=_db.Open();using var q=c.CreateCommand();q.CommandText="PRAGMA query_only=ON;";q.ExecuteNonQuery();return c;
        }
    }
    private sealed class Broken:ISaveDatabase
    {
        public SqliteConnection Open()=>throw new UnauthorizedAccessException("synthetic");
    }
    private sealed class F:IDisposable
    {
        internal readonly SyntheticSandbox Sandbox=new(SyntheticSandbox.AuthorizedProjectRoot);
        internal readonly SyntheticDatabase Db;
        internal readonly GameSaveService Save;
        internal readonly string Owner=Id();
        internal string Game="";
        internal F(bool profile=true)
        {
            Db=Sandbox.CreateDatabase();SchemaManager.Initialize(Db);Save=new(Db,RealGameCatalog.InitialState());
            if(profile)Save.CreateLocalProfile(Id(),Owner,"Estadísticas sintéticas");
        }
        internal void NewGame(){Game=Id();Save.CreateNewGame(new(Id(),Owner,Game));}
        internal void Decide(int index,bool healthy=true)
        {
            var g=Save.LoadGame(Owner,Game).Game;Save.EnterBiome(new(Id(),Owner,Game,g.Revision,Biomes[index],"human"));
            g=Save.LoadGame(Owner,Game).Game;Save.RecordDecision(new(Id(),Id(),Owner,Game,g.Revision,Biomes[index],Keys[index],healthy?"reject":"approve",healthy?"healthy":"degraded",healthy?"{\"stateKind\":\"healthy\"}":"{\"stateKind\":\"degraded\"}"));
        }
        internal void Terminal(string ending="good")
        {
            // Fixture setup, not a production implementation of finalization.
            using var c=Db.Open();using var q=c.CreateCommand();
            q.CommandText="UPDATE saved_games SET is_current=0,status='completed',ending=$e,ending_rules_version=rules_version,completed_at=saved_at WHERE game_id=$g;";
            q.Parameters.AddWithValue("$e",ending);q.Parameters.AddWithValue("$g",Game);q.ExecuteNonQuery();
        }
        internal StatisticsService Stats()=>new(new ReadOnly(Db),RealGameCatalog.InitialState());
        internal string Hash()=>Convert.ToHexString(SHA256.HashData(File.ReadAllBytes(Db.Path)));
        internal void Sql(string sql){using var c=Db.Open();using var q=c.CreateCommand();q.CommandText=sql;q.ExecuteNonQuery();}
        public void Dispose()=>Sandbox.Dispose();
    }
    private static async Task Case(string name,Func<F,Task> work,bool profile=true)
    {using var f=new F(profile);await work(f);_count++;Console.WriteLine("PASS "+name);}
    public static async Task<int> Run()
    {
        await Case("initialized empty database returns NoData without creating profile",f=>
        {string h=f.Hash();var r=f.Stats().GetLocalStatistics();Check(r.State==StatisticsAvailability.NoData && r.Data is null && f.Hash()==h);return Task.CompletedTask;},false);
        await Case("profile without games has measured zero decisions and no history",f=>
        {var p=f.Stats().GetPlayerStatistics(f.Owner);Check(p.LocalPlayerId==f.Owner && p.DisplayName=="Estadísticas sintéticas" && p.TotalGames==0 && p.AllDecisions.Total==0 && p.GamesState==StatisticsAvailability.NoData && p.HistoricalGames.Count==0);return Task.CompletedTask;});
        await Case("one active game is separate from historical games",f=>
        {f.NewGame();var p=f.Stats().GetPlayerStatistics(f.Owner);Check(p.TotalGames==1 && p.ActiveGames==1 && p.ActiveGameId==f.Game && p.ActiveGame?.GameId==f.Game && p.HistoricalGames.Count==0);return Task.CompletedTask;});
        await Case("five precreated human pending rows do not mean five started biomes",f=>
        {f.NewGame();var g=f.Stats().GetGameStatistics(f.Owner,f.Game);Check(g.Progress.Pending==5 && g.Progress.Completed==0 && g.Progress.ConfirmedStartedMinimum==0 && g.Progress.Started.State==StatisticsAvailability.Unknown && g.Progress.Started.Value is null);return Task.CompletedTask;});
        await Case("current human entry is explicit started evidence",f=>
        {f.NewGame();f.Save.EnterBiome(new(Id(),f.Owner,f.Game,0,"desert","human"));var g=f.Stats().GetGameStatistics(f.Owner,f.Game);Check(g.Progress.ConfirmedStartedMinimum==1 && g.Biomes.Single(b=>b.BiomeId=="desert").StartedEvidence=="CURRENT_HUMAN_ENTRY");return Task.CompletedTask;});
        for(int i=0;i<5;i++)foreach(bool healthy in new[]{true,false})
        {
            int index=i;
            await Case("confirmed decision "+Biomes[i]+" "+(healthy?"reject":"approve"),f=>
            {
                f.NewGame();f.Decide(index,healthy);var g=f.Stats().GetGameStatistics(f.Owner,f.Game);
                var b=g.Biomes.Single(b=>b.BiomeId==Biomes[index]);
                Check(g.Decisions.Total==1 && g.Decisions.Reject==(healthy?1:0) && g.Decisions.Approve==(healthy?0:1) && g.Decisions.Healthy==(healthy?1:0) && g.Decisions.Degraded==(healthy?0:1));
                Check(b.Consequence?.StateKind==(healthy?"healthy":"degraded") && b.Consequence.GameId==f.Game && b.Consequence.OperationId==f.Save.LoadGame(f.Owner,f.Game).Decisions.Single().OperationId);
                Check(g.Progress.Completed==0 && g.Progress.Pending==5 && b.Status=="animal_pending" && b.Started==true);
                return Task.CompletedTask;
            });
        }
        await Case("free biome order retains each option and scope",f=>
        {f.NewGame();foreach(int i in new[]{4,0,3,1,2})f.Decide(i,i%2==0);var g=f.Stats().GetGameStatistics(f.Owner,f.Game);Check(g.Decisions.Total==5 && g.Decisions.Reject==3 && g.Decisions.Approve==2 && g.Progress.Started.Value==5 && g.Decisions.ByBiome.All(b=>b.Total==1));return Task.CompletedTask;});
        await Case("idempotent decision retry and repeated queries do not duplicate counts",f=>
        {
            f.NewGame();f.Save.EnterBiome(new(Id(),f.Owner,f.Game,0,"ocean","human"));
            var command=new RecordDecisionCommand(Id(),Id(),f.Owner,f.Game,1,"ocean",Keys[0],"reject","healthy","{\"stateKind\":\"healthy\"}");
            f.Save.RecordDecision(command);string h=f.Hash();f.Save.RecordDecision(command);
            for(int i=0;i<3;i++)Check(f.Stats().GetPlayerStatistics(f.Owner).AllDecisions.Total==1);
            Check(f.Hash()==h);return Task.CompletedTask;
        });
        await Case("animal restart query does not increase decisions",async f=>
        {
            f.NewGame();f.Decide(0);var store=new RuntimeMenuChecks.Store(f.Db,new RuntimeMenuChecks.Failure());
            await using var state=new LocalGameState(store,"owned-fixture");await state.InitializeAsync();string h=f.Hash();

        });
        await Case("multiple historical games and active decisions stay separate",f=>
        {
            f.NewGame();f.Decide(0);string first=f.Game;f.Terminal("medium");f.NewGame();f.Decide(1,false);

        });

        {

            return Task.CompletedTask;
        });
        await Case("repository pagination continues past its own 128-row page",f=>
        {for(int n=0;n<129;n++){f.NewGame();f.Terminal();}var p=f.Stats().GetPlayerStatistics(f.Owner);Check(p.TotalGames==129 && p.CompletedGames.Value==129 && p.HistoricalGames.Count==129);return Task.CompletedTask;});

        {

        });
        await Case("terminal metadata is read without evaluating a new ending",f=>
        {f.NewGame();f.Terminal("bad");var g=f.Stats().GetGameStatistics(f.Owner,f.Game);Check(g.ConfirmedCompleted && g.Ending.Kind=="bad" && g.Ending.State==StatisticsAvailability.Available && g.CompletedAt is not null);return Task.CompletedTask;});

        await Case("current final-pending record stays separate from terminal history",f=>
        {f.NewGame();f.Sql("UPDATE saved_games SET status='final_pending',ending='medium',ending_rules_version=rules_version,completed_at=saved_at,resume_kind='ending';");var p=f.Stats().GetPlayerStatistics(f.Owner);Check(p.ActiveGames==1 && p.ActiveGame?.Status=="final_pending" && p.ActiveGame.Ending.Kind=="medium" && p.CompletedGames.Value==0 && p.HistoricalGames.Count==0);return Task.CompletedTask;});

        {var p=f.Stats().GetPlayerStatistics(f.Owner);foreach(var m in new[]{p.Uncollected.PlayedMilliseconds,p.Uncollected.Deaths,p.Uncollected.Restarts,p.Uncollected.Sessions,p.Uncollected.Retries})Check(m.State==StatisticsAvailability.NotAvailable && m.Value is null);Check(p.AllDecisions.Total==0 && p.CompletedGames.Value==0);return Task.CompletedTask;});
