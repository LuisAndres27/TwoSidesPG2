    // Child-process capability is restricted to fresh synthetic UUID paths, never runtime/AppData.
    internal sealed class ProbeStore:ISaveDatabase
    {
        private readonly string _path;
        internal ProbeStore(string path)
        {
            _path=Path.GetFullPath(path);
            string parent=Path.GetDirectoryName(_path)!;
            string expected=Path.Combine(SyntheticSandbox.AuthorizedProjectRoot,".work","databases");
            SaveValidation.Require(Path.GetDirectoryName(parent)==expected &&
                Guid.TryParseExact(Path.GetFileName(parent),"N",out _) &&
                Guid.TryParseExact(Path.GetFileNameWithoutExtension(_path),"N",out _) && Path.GetExtension(_path)==".sqlite3","UNSAFE_PATH");
            foreach(string p in new[]{SyntheticSandbox.AuthorizedProjectRoot,Path.Combine(SyntheticSandbox.AuthorizedProjectRoot,".work"),expected,parent,_path})
                SaveValidation.Require((File.GetAttributes(p)&FileAttributes.ReparsePoint)==0,"UNSAFE_PATH");
        }
        public SqliteConnection Open()
        {
            var c=new SqliteConnection(new SqliteConnectionStringBuilder{DataSource=_path,Mode=SqliteOpenMode.ReadWrite,Pooling=false,ForeignKeys=true,DefaultTimeout=1}.ToString());
            c.Open();using var q=c.CreateCommand();q.CommandText="PRAGMA synchronous=FULL;PRAGMA busy_timeout=1000;";q.ExecuteNonQuery();return c;
        }
    }
    public static int Probe(string[] args)
    {
        // mode,path,owner,installation,operation,time
        var db=new ProbeStore(args[1]);string owner=args[2],installation=args[3],op=args[4];
        var clock=DateTimeOffset.FromUnixTimeMilliseconds(long.Parse(args[5],CultureInfo.InvariantCulture));
        var binding=new SyncBinding(owner,installation,1);
        var queue=new SyncOutboxService(db,()=>clock,()=>0);
        if(args[0]=="read")return queue.Get(owner,op)?.State==OutboxDeliveryState.Pending?0:81;
        if(args[0]=="inflight")return queue.Get(owner,op)?.State==OutboxDeliveryState.InFlight?0:82;
        if(args[0]=="recover")return queue.RecoverExpired(binding)==1?0:83;
        if(args[0]=="crash-reserve"){new SyncOutboxService(db,()=>clock,()=>0,()=>Environment.Exit(73)).Reserve(binding);return 84;}
        if(args[0]=="crash-confirm")
        {
            var row=queue.Get(owner,op)!;
            new SyncOutboxService(db,()=>clock,()=>0,()=>Environment.Exit(73)).Confirm(new(op,owner,installation,1,row.PayloadHash,row.LeaseId!,row.Attempts,Id(),"2026-10-27T12:00:00.000Z"));
            return 85;
        }
        return 86;
    }
    private static int Child(F f,string mode,string op)
    {
        var start=new ProcessStartInfo(Environment.ProcessPath!){UseShellExecute=false,CreateNoWindow=true,RedirectStandardOutput=true,RedirectStandardError=true};
        foreach(string arg in new[]{"--outbox-probe",mode,f.Db.Path,f.Owner,f.Installation,op,f.Clock.ToUnixTimeMilliseconds().ToString(CultureInfo.InvariantCulture)})start.ArgumentList.Add(arg);
        using var child=Process.Start(start)!;
        if(!child.WaitForExit(15000)){child.Kill();throw new Exception("Synthetic child timeout");}
        Check(child.StandardOutput.ReadToEnd().Length==0 && child.StandardError.ReadToEnd().Length==0);
        return child.ExitCode;
    }
