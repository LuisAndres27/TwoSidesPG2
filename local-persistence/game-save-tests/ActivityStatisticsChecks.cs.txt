        internal void Interval(long start,long end,GameplayTimeState state=GameplayTimeState.ActiveAnimal)=>Events.RecordInterval(new(Id(),Owner,Session,start,end,state,Game,Biome,Run));
        internal void Close(long offset)=>Events.RecordEvent(new(Id(),Owner,Session,GameplayEventKind.SESSION_ENDED,DateTimeOffset.UnixEpoch.AddMilliseconds(offset),offset,SessionEnd:GameplaySessionEnd.Normal));

        internal string Hash()=>Convert.ToHexString(SHA256.HashData(File.ReadAllBytes(Db.Path)));
        public void Dispose()=>Sandbox.Dispose();
    }
    private static async Task Case(string name,Func<F,Task> work,bool extension=true)
    {using var f=new F(extension);await work(f);_count++;Console.WriteLine("PASS "+name);}
    public static async Task<int> Run()
    {
        await Case("base-only history remains NotAvailable not measured zero",f=>
        {Check(f.Activity().Sessions.State==StatisticsAvailability.NotAvailable && f.Activity().EffectiveMs.Value is null && f.Player().AllDecisions.Total==5);return Task.CompletedTask;},false);
        await Case("migrated history before capture has no coverage rather than zero deaths",f=>
        {var a=f.Activity();Check(a.Sessions.State==StatisticsAvailability.NoData && a.Deaths.Value is null && a.EffectiveMs.Value is null && a.HistoricalCoverageUnknown);return Task.CompletedTask;});

        {f.Start();f.Animal();f.Interval(0,100);f.Close(100);var a=f.Activity();Check(a.Sessions.Value==1 && a.EffectiveMs.Value==100 && !a.Incomplete);return Task.CompletedTask;});

        await Case("confirmed death differs from retry and voluntary request",f=>
        {f.Start();f.Animal();f.Events.RecordEvent(f.Event(GameplayEventKind.ANIMAL_DIED,10));f.Events.RecordEvent(f.Event(GameplayEventKind.ANIMAL_RETRY_REQUESTED,20));f.Animal(0,20);f.Events.RecordEvent(f.Event(GameplayEventKind.ANIMAL_RESTART_REQUESTED,30));var a=f.Activity();Check(a.Deaths.Value==1 && a.RetryRequests.Value==1 && a.RestartRequests.Value==1);return Task.CompletedTask;});
        await Case("idempotent death is counted once in the visible aggregate",f=>
        {f.Start();f.Animal();var death=f.Event(GameplayEventKind.ANIMAL_DIED,10);f.Events.RecordEvent(death);f.Events.RecordEvent(death);Check(f.Activity().Deaths.Value==1);return Task.CompletedTask;});
        await Case("pause menu background Game Over and loading excluded",f=>
        {f.Start();f.Animal();long offset=0;foreach(var state in new[]{GameplayTimeState.ActiveAnimal,GameplayTimeState.Paused,GameplayTimeState.Menu,GameplayTimeState.GameOver,GameplayTimeState.Loading,GameplayTimeState.Background}){f.Interval(offset,offset+100,state);offset+=100;}Check(f.Activity().EffectiveMs.Value==100);return Task.CompletedTask;});

        {int index=i;await Case("canonical biome subtotal "+Biomes[i],f=>
        {f.Start();f.Animal(index);f.Interval(0,100+index);var a=f.Activity();Check(a.ByBiome.Single(b=>b.BiomeId==Biomes[index]).EffectiveMs.Value==100+index && a.ByBiome.Count(b=>b.EffectiveMs.Value is null)==4);return Task.CompletedTask;});}
        await Case("all five biome totals sum to effective total",f=>
        {f.Start();for(int i=0;i<5;i++){f.Animal(i,i*100);f.Interval(i*100,(i+1)*100);f.Events.RecordEvent(f.Event(GameplayEventKind.ANIMAL_SEGMENT_EXITED,(i+1)*100));}var a=f.Activity();Check(a.EffectiveMs.Value==500 && a.ByBiome.Sum(b=>b.EffectiveMs.Value??0)==500 && a.Sessions.Value==1);return Task.CompletedTask;});
        await Case("measured zero is distinct from an unobserved biome",f=>
        {f.Start();f.Animal();f.Interval(0,0);var a=f.Activity();Check(a.EffectiveMs.State==StatisticsAvailability.Available && a.EffectiveMs.Value==0 && a.ByBiome.Single(b=>b.BiomeId=="forest").EffectiveMs.State==StatisticsAvailability.NoData);return Task.CompletedTask;});
        await Case("session alone does not invent zero animal deaths",f=>
        {f.Start();Check(f.Activity().Sessions.Value==1 && f.Activity().Deaths.Value is null && f.Activity().EffectiveMs.Value is null);return Task.CompletedTask;});
        await Case("active game and history activity remain separate",f=>
        {
            f.Start();f.Animal();f.Interval(0,100);f.Events.RecordEvent(f.Event(GameplayEventKind.ANIMAL_SEGMENT_EXITED,100));f.Close(100);
