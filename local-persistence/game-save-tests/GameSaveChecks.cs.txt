internal static class BiomeProgressChecks
{
    private static int _passed,_failed;
    private static string Id()=>Guid.NewGuid().ToString("D");
    private static void Require(bool value){if(!value)throw new Exception("Assertion failed");}
    private static string Error(Action action)
    {
        try{action();return "OK";}catch(SaveException e){return e.Code;}
    }
    private sealed class UnknownPolicy:IBiomeAvailabilityPolicy { public BiomeAvailability Evaluate(GameAggregate game,string biome)=>BiomeAvailability.Unknown; }
    private sealed class FixturePolicy:IBiomeAvailabilityPolicy
    {
        public BiomeAvailability Evaluate(GameAggregate game,string biome)=>biome switch
        {"ocean"=>BiomeAvailability.Available,"jungle"=>BiomeAvailability.Unavailable,_=>BiomeAvailability.Unknown};
    }
    private sealed class RejectCommit:ITransactionalSaveExtension
    {
        public void BeforeCommit(SqliteConnection c,SqliteTransaction t,SaveReceipt r)=>throw new SaveException("TEST_ROLLBACK");
    }
    private sealed class FailingSqlWrite:ITransactionalSaveExtension
    {
        public void BeforeCommit(SqliteConnection c,SqliteTransaction t,SaveReceipt receipt)
        {
            using var cmd=c.CreateCommand();cmd.Transaction=t;
            cmd.CommandText="INSERT INTO biome_progress VALUES($g,'ocean','human_pending',NULL,0,NULL,NULL,'fixture');";
            cmd.Parameters.AddWithValue("$g",Id());cmd.ExecuteNonQuery();
        }
    }
    private sealed class F:IDisposable
    {
        internal readonly SyntheticSandbox Sandbox=new(SyntheticSandbox.AuthorizedProjectRoot);
        internal readonly SyntheticDatabase Db;
        internal readonly GameSaveService S;
        internal readonly string Owner=Id();
        internal string Game;
        internal F()
        {
            Db=Sandbox.CreateDatabase();SchemaManager.Initialize(Db);
            S=new GameSaveService(Db,RealGameCatalog.InitialState(),availability:new FixturePolicy());
            S.CreateLocalProfile(Id(),Owner,"Fixture progreso");Game=Id();S.CreateNewGame(new(Id(),Owner,Game));
        }
        internal long Revision=>S.LoadGame(Owner,Game).Game.Revision;
        internal RecordDecisionCommand Decision(string biome="ocean",string option="reject",EnvironmentalClassification? classification=null)=>
            new(Id(),Id(),Owner,Game,Revision,biome,biome=="ocean"?"ocean_waste_disposal":"jungle_transport_route",option,
                "healthy","{\"stateKind\":\"healthy\"}",classification);
        internal CompleteBiomeCommand Completion()=>new(Id(),Owner,Game,Revision,"ocean","good");
        internal EnterBiomeCommand Entry(string biome="ocean",string perspective="human")=>new(Id(),Owner,Game,Revision,biome,perspective);
        internal string State()=>System.Text.Json.JsonSerializer.Serialize(S.LoadGame(Owner,Game));
        internal void Finish(){S.RecordDecision(Decision());S.CompleteBiome(Completion());}
        public void Dispose()=>Sandbox.Dispose();
    }
    private static void Case(string name,Action<F> test)
    {
        try{using var f=new F();test(f);_passed++;Console.WriteLine("PASS "+name);}
        catch(Exception e){_failed++;Console.WriteLine("FAIL "+name+": "+(e is SaveException s?s.Code:e.GetType().Name));}
    }
    private static async Task AsyncCase(string name,Func<F,Task> test)
    {
        try{using var f=new F();await test(f);_passed++;Console.WriteLine("PASS "+name);}
        catch(Exception e){_failed++;Console.WriteLine("FAIL "+name+": "+(e is SaveException s?s.Code:e.GetType().Name));}
    }
    private sealed class ReadOnlyFixture:ISaveDatabase
    {
        private readonly string _path;
        internal ReadOnlyFixture(string path)
        {
            string root=System.IO.Path.Combine(SyntheticSandbox.AuthorizedProjectRoot,".work","databases");
            string parent=System.IO.Path.GetDirectoryName(path)!;
            Require(System.IO.Path.IsPathFullyQualified(path)&&System.IO.Path.GetDirectoryName(parent)==root
                &&Guid.TryParseExact(System.IO.Path.GetFileName(parent),"N",out _)
                &&Guid.TryParseExact(System.IO.Path.GetFileNameWithoutExtension(path),"N",out _)
                &&System.IO.Path.GetExtension(path)==".sqlite3");
            foreach(string p in new[]{SyntheticSandbox.AuthorizedProjectRoot,System.IO.Path.Combine(SyntheticSandbox.AuthorizedProjectRoot,".work"),root,parent,path})
                Require((File.GetAttributes(p)&FileAttributes.ReparsePoint)==0);
            _path=path;
        }
        public SqliteConnection Open()
        {
            var c=new SqliteConnection(new SqliteConnectionStringBuilder{DataSource=_path,Mode=SqliteOpenMode.ReadOnly,Pooling=false,ForeignKeys=true}.ToString());c.Open();return c;
        }
    }
