        {

            {

            }
            tx.Commit();return string.Join("\n",result);
        }

    public static int Probe(string[] args)
    {
        // mode,synthetic path,owner,installation. Never an arbitrary runtime capability.
        var db=new OutboxChecks.ProbeStore(args[1]);var binding=new SyncBinding(args[2],args[3],1);
        SyncReconciliationCursor? cursor=null;
        try
        {
            do
            {
                using var c=db.Open();using var tx=c.BeginTransaction(deferred:false);
                var page=SyntheticOutboxReconciler.PrepareSyntheticPage(new(c,tx),binding,Time,10,cursor);
                if(args[0]=="crash")Environment.Exit(74);
                if(page.Diagnostics.Count!=0)return 79;
                tx.Commit();cursor=page.Next;
            }while(cursor is not null);
            return 0;
        }
        catch(SqliteException e) when(e.SqliteErrorCode is 5 or 6){return 75;}
    }
    internal static async Task<int> Child(F f,string mode)
    {
        var psi=new ProcessStartInfo(Environment.ProcessPath!){UseShellExecute=false,CreateNoWindow=true,RedirectStandardOutput=true,RedirectStandardError=true};
        foreach(string arg in new[]{"--reconcile-probe",mode,f.Db.Path,f.Owner,f.Installation})psi.ArgumentList.Add(arg);
        using var process=Process.Start(psi)!;using var timeout=new System.Threading.CancellationTokenSource(TimeSpan.FromSeconds(20));
        try{await process.WaitForExitAsync(timeout.Token);}catch{process.Kill();throw new Exception("Synthetic process timeout");}
        Check((await process.StandardOutput.ReadToEndAsync()).Length==0 && (await process.StandardError.ReadToEndAsync()).Length==0);
        return process.ExitCode;
    }
