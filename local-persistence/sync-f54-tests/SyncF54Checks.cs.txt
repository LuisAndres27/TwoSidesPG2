// B54: owner-quota handling of the synchronization client, WITHOUT any server process. The HTTP side is an in-process HttpMessageHandler that answers exactly like the real
// Sync v2 service does for an exhausted owner quota (batch answer with per-operation error code "quota_exceeded") and for the rate limiter (error envelope, no "results").
// Databases are disposable synthetic SQLite. No Express, no Caddy, no Atlas, no real saves.
internal static class SyncF54Checks

    private static readonly DateTimeOffset Start=DateTimeOffset.Parse("2026-10-27T12:00:00.000Z",CultureInfo.InvariantCulture);
    private static string Id()=>Guid.NewGuid().ToString("D");
    private static void Check(bool ok,string label="check"){if(!ok)throw new Exception("B54 assertion failed: "+label);}

    private sealed class F:IDisposable
    {
        internal readonly SyntheticSandbox Sandbox=new(SyntheticSandbox.AuthorizedProjectRoot);
        internal readonly SyntheticDatabase Db;
        internal readonly string Owner=Id(),Game=Id(),Installation=Id();
        internal DateTimeOffset Clock=Start;
        internal SyncBinding Binding=>new(Owner,Installation,1);
        internal SyncOutboxService Queue=>new(Db,()=>Clock,()=>0);
        internal F()
        {
            Db=Sandbox.CreateDatabase();SchemaManager.Initialize(Db);var save=new GameSaveService(Db,RealGameCatalog.InitialState(),()=>Start);
            save.CreateLocalProfile(Id(),Owner,"Synthetic B54");save.CreateNewGame(new(Id(),Owner,Game));
            GameplayEventSchema.MigrateSynthetic(Db);SyncOutboxSchema.MigrateSynthetic(Db);Queue.RegisterInstallation(Binding);
        }
        // n sessions; returns the open-report ids (independent operations).
        internal List<string> Sessions(int n,bool closedToo=false,List<string>? closedIds=null)
        {
            var ids=new List<string>();var events=new GameplayEventService(Db);
            for(int i=0;i<n;i++)
            {
                string session=Id();
                events.RecordEvent(new(Id(),Owner,session,GameplayEventKind.SESSION_STARTED,Start,0));
                events.RecordEvent(new(Id(),Owner,session,GameplayEventKind.SESSION_ENDED,Start.AddSeconds(1),1000,SessionEnd:GameplaySessionEnd.Normal));
                string open=Queue.Capture(new(Id(),Binding,new(SyncFactKind.SessionReport,session,Phase:"open"))).Operation.OperationId;ids.Add(open);
                if(closedToo)closedIds?.Add(Queue.Capture(new(Id(),Binding,new(SyncFactKind.SessionReport,session,Phase:"terminal"),new[]{open})).Operation.OperationId);
            }
            return ids;
        }
        internal long Number(string sql){using var c=Db.Open();using var q=c.CreateCommand();q.CommandText=sql;return Convert.ToInt64(q.ExecuteScalar());}
        internal string Text(string sql){using var c=Db.Open();using var q=c.CreateCommand();q.CommandText=sql;return Convert.ToString(q.ExecuteScalar(),CultureInfo.InvariantCulture)??"";}
        internal int State(string state)=>(int)Number("SELECT count(*) FROM sync_outbox WHERE state='"+state+"';");
        internal long MinNextAttempt(string where)=>Number("SELECT COALESCE(min(next_attempt_ms),0) FROM sync_outbox WHERE "+where+";");
        // Everything that must never change: identifiers, payloads, hashes, source keys, creation order and the dependency graph.
        internal string Fingerprint()=>Text("SELECT group_concat(operation_id||'|'||payload_hash||'|'||source_key||'|'||created_at_utc||'|'||type||'|'||length(canonical_envelope),';') FROM (SELECT * FROM sync_outbox ORDER BY operation_id)")
            +"#"+Text("SELECT group_concat(operation_id||'>'||depends_on_id,';') FROM (SELECT * FROM sync_dependencies ORDER BY operation_id,depends_on_id)");
        internal long NowMs=>Clock.ToUnixTimeMilliseconds();
        public void Dispose()=>Sandbox.Dispose();
    }

    private sealed class Ticks{internal long Ms=0;}
    internal enum Mode{Normal,RateLimiter,CommitThenDrop}
    private sealed record Request(string Path,JsonElement Body);

    private sealed class FakeServer:HttpMessageHandler
    {
        private readonly SyncBinding _binding;
        internal Mode Behaviour=Mode.Normal;
        internal int Capacity=int.MaxValue;                                   // new operations the owner may still store
        internal readonly List<Request> Requests=new();
        internal readonly Dictionary<string,JsonElement> Receipts=new();
        internal readonly List<string> Committed=new();
        internal FakeServer(SyncBinding binding){_binding=binding;}
        internal IEnumerable<Request> Batches=>Requests.Where(r=>r.Path.EndsWith("/sync/batches",StringComparison.Ordinal));
        internal static string[] OperationIds(Request r)=>r.Body.GetProperty("operations").EnumerateArray().Select(o=>o.GetProperty("operationId").GetString()!).ToArray();
        private static HttpResponseMessage Json(HttpStatusCode status,object body)=>new(status){Content=new StringContent(JsonSerializer.Serialize(body),new UTF8Encoding(false),"application/json")};
        private JsonElement Receipt(JsonElement op)
        {
            string id=op.GetProperty("operationId").GetString()!;
            var receipt=JsonSerializer.SerializeToElement(new{receiptVersion=1,operationContractVersion=1,technicalAccountId=Id(),operationId=id,localPlayerId=_binding.LocalPlayerId,
                installationId=_binding.InstallationId,bindingGeneration=1,payloadHash=op.GetProperty("payloadHash").GetString(),receiptId=Id(),committedAtUtc="2026-10-27T12:00:00.000Z",acceptedAs="committed"});
            Receipts[id]=receipt;Committed.Add(id);return receipt;
        }
        protected override async Task<HttpResponseMessage> SendAsync(HttpRequestMessage request,CancellationToken cancellation)
        {
            string path=request.RequestUri!.AbsolutePath;
            JsonElement body=default;
            if(request.Content is not null){var bytes=await request.Content.ReadAsByteArrayAsync(cancellation);using var doc=JsonDocument.Parse(bytes);body=doc.RootElement.Clone();}
            Requests.Add(new(path,body));
            if(path.EndsWith("/technical/registrations",StringComparison.Ordinal))return Json(HttpStatusCode.Created,new{schemaVersion=2,technicalAccountId=Id()});
            if(path.EndsWith("/technical/sessions",StringComparison.Ordinal))return Json(HttpStatusCode.OK,new{schemaVersion=2,accessToken="synthetic-token",expiresIn=900});
            if(path.EndsWith("/sync/receipts/query",StringComparison.Ordinal))
            {
                var results=body.GetProperty("attempts").EnumerateArray().Select(a=>
                {
                    string id=a.GetProperty("operationId").GetString()!;
                    return Receipts.TryGetValue(id,out var r)?(object)new{operationId=id,status="found",receipt=r,attempt=new{leaseId=a.GetProperty("leaseId").GetString(),attempt=a.GetProperty("attempt").GetInt32()}}
                                                              :new{operationId=id,status="not_found"};
                }).ToArray();
                return Json(HttpStatusCode.OK,new{schemaVersion=2,requestId=Id(),results});
            }
            if(path.EndsWith("/sync/batches",StringComparison.Ordinal))
            {
                var ops=body.GetProperty("operations").EnumerateArray().ToArray();
                if(Behaviour==Mode.RateLimiter)
                {   // the rate limiter answers from the router error handler: error envelope, NO results[]
                    var limited=new HttpResponseMessage((HttpStatusCode)429){Content=new StringContent(JsonSerializer.Serialize(new{schemaVersion=2,error=new{code="rate_limited",message="Sync request unavailable.",retryable=true},requestId=Id()}),new UTF8Encoding(false),"application/json")};
                    limited.Headers.RetryAfter=new System.Net.Http.Headers.RetryConditionHeaderValue(TimeSpan.FromSeconds(1));return limited;
                }
                if(Behaviour==Mode.CommitThenDrop){foreach(var op in ops)if(Capacity>0){Capacity--;Receipt(op);}throw new HttpRequestException("response lost after commit");}
                var list=new List<object>();int accepted=0;
                for(int i=0;i<ops.Length;i++)
                {
                    var o=ops[i];string id=o.GetProperty("operationId").GetString()!;
                    var attempt=new{leaseId=o.GetProperty("leaseId").GetString(),attempt=o.GetProperty("attempt").GetInt32()};
                    if(Receipts.TryGetValue(id,out var again)){accepted++;list.Add(new{index=i,operationId=id,status="duplicate",httpStatus=200,receipt=again,attempt});continue;}   // duplicates are answered BEFORE the quota, as the real engine does
                    if(Capacity<=0){list.Add(new{index=i,operationId=id,status="deferred",httpStatus=429,error=new{code="quota_exceeded",retryable=true,retryAfterSeconds=1}});continue;}
                    Capacity--;accepted++;list.Add(new{index=i,operationId=id,status="committed",httpStatus=201,receipt=Receipt(o),attempt});
                }
                var status=accepted==ops.Length?HttpStatusCode.OK:accepted>0?(HttpStatusCode)207:(HttpStatusCode)429;
                return Json(status,new{schemaVersion=2,batchId=body.GetProperty("batchId").GetString(),requestId=Id(),outcome=accepted==ops.Length?"all_accepted":accepted>0?"partial":"none_accepted",results=list});
            }
            return new HttpResponseMessage(HttpStatusCode.NotFound);
        }
    }

    private sealed class Client:IDisposable
    {
        internal readonly Ticks Time=new();
        internal readonly FakeServer Server;
        internal readonly SyncHttpTransport Transport;
        internal readonly SyncSessionManager Session;
        internal readonly SyncOutboxDispatcher Dispatcher;
        // `server` lets a "restarted game" (a new Client) talk to the same fake server.
        internal Client(F f,FakeServer? server=null,TimeSpan[]? pauses=null)
        {
            Server=server??new FakeServer(f.Binding);
            var options=new SyncHttpOptions(new Uri("http://127.0.0.1:9/"),null,allowLoopbackHttp:true);
            Transport=new(options,Server);
            Session=new(Transport,f.Binding,new InMemorySyncSecretStore());
            Dispatcher=new(f.Queue,Session,f.Binding,SyncHttpOptions.InitialBatchOperations,SyncHttpOptions.MaxBatchOperationsLimit,()=>Time.Ms,pauses);
        }
        internal Task<SyncPassSummary> Pass()=>Dispatcher.RunPassAsync(CancellationToken.None);
        public void Dispose()=>Transport.Dispose();
    }

    private static async Task Case(string label,Func<F,Task> body){using var f=new F();await body(f);_count++;Console.WriteLine("PASS "+label);}
    private static async Task Case(string label,Action<F> body){using var f=new F();body(f);_count++;Console.WriteLine("PASS "+label);await Task.CompletedTask;}

                {
                    var s=await dispatcher.RunPassAsync(CancellationToken.None);last=s.Code;
                    if(s.Code=="SYNC_QUOTA_BLOCKED")break;
                    if(s.Code=="SYNC_RATELIMITED"){rateLimited++;f.Clock=f.Clock.AddSeconds(30);await Task.Delay(1500);}
                    else if(s.Code.StartsWith("SYNC_STORAGE_",StringComparison.Ordinal) || s.Code is "SYNC_NETWORKUNAVAILABLE" or "SYNC_TIMEOUT"){throw new Exception("unexpected "+s.DiagnosticSummary);}
                }
                Console.WriteLine($"  real server: passes={passes} rateLimitedPasses={rateLimited} last={last} confirmed={f.State("confirmed")} retryable={f.State("retryable_failure")} pending={f.State("pending")}");
                Check(last=="SYNC_QUOTA_BLOCKED","blocked by the real quota");
                // The in-memory offline-test store also has a 2 MiB valve (ingestion-engine.js:121, answered with the SAME quota_exceeded code), which it reaches before 1000 operations; the Mongo-backed service uses the count only.
                int accepted=f.State("confirmed");
                Check(accepted>=500 && accepted<=1000,"accepted up to the server limit: "+accepted);
                Check(f.State("permanent_failure")==0 && f.State("confirmed")+f.State("pending")+f.State("retryable_failure")==total,"nothing lost, nothing permanent");
                Check(f.Number("SELECT count(*) FROM sync_outbox WHERE state<>'confirmed' AND next_attempt_ms>="+(f.NowMs+14*60*1000))==total-accepted,"every unconfirmed operation is paused");
                Check(f.Fingerprint()==before,"identifiers and payloads unchanged");
                int again=0;for(int i=0;i<30;i++){var s=await dispatcher.RunPassAsync(CancellationToken.None);if(s.Reserved==0 && s.Code=="SYNC_QUOTA_BLOCKED")again++;}
                Check(again==30,"30 further passes: no reservation, no request");
            });

        if(express is not null && which=="rate")
            await Case("REAL Express offline-test: the real rate limiter (120 requests/min per owner) answers a different 429 shape and is NOT treated as a quota block",async f=>{
                f.Sessions(160);var options=new SyncHttpOptions(new Uri(express),null,allowLoopbackHttp:true);
                using var transport=new SyncHttpTransport(options);var session=new SyncSessionManager(transport,f.Binding,new InMemorySyncSecretStore());
                var dispatcher=new SyncOutboxDispatcher(f.Queue,session,f.Binding,1,1);
                string last="";int limited=0;
                for(int i=0;i<200 && limited==0;i++){var s=await dispatcher.RunPassAsync(CancellationToken.None);last=s.Code;if(s.Code=="SYNC_RATELIMITED")limited++;}
                Console.WriteLine("  real server: last="+last+" confirmed="+f.State("confirmed")+" rate_limited rows="+f.Number("SELECT count(*) FROM sync_outbox WHERE last_error_code='rate_limited'"));
                Check(limited==1 && last=="SYNC_RATELIMITED","the real limiter was reached and reported as rate limiting");
                Check(f.Queue.QuotaPauseUntil(f.Binding) is null && !dispatcher.QuotaProbePending,"no quota state");
                Check(f.Number("SELECT count(*) FROM sync_outbox WHERE last_error_code='quota_exceeded'")==0 && f.State("permanent_failure")==0,"no quota code, no permanent failure");
            });

        Console.WriteLine($"B54 quota checks: {_count} cases passed.");
        return 0;
