 static void FinalsAndOutbox(){
  for(int rejects=0;rejects<=5;rejects++)using(var f=new F()){
   f.Start();Check(f.N("SELECT count(*) FROM statistics_events WHERE event_type='installation_registered'")==1,"single installation registration");
   Check(f.N("SELECT count(*) FROM statistics_events WHERE event_type='game_started'")==1,"game start captured");

   Check(f.N("SELECT count(*) FROM statistics_events WHERE event_type='environmental_decision'")==5,"only confirmed decisions, retry deduplicated");
   Check(f.N("SELECT count(*) FROM statistics_events WHERE event_type='biome_completed'")==5,"five physical completions");
   var final=new PrepareFinalCommand(Id(),f.Owner,f.Game,f.Revision);f.Save.PrepareFinal(final);f.Save.PrepareFinal(new(Id(),f.Owner,f.Game,f.Revision));
   Check(f.N("SELECT count(*) FROM statistics_events WHERE event_type='game_completed'")==1,"valid ending captured once across duplicate intents");
   var expected=rejects==5?"good":rejects>=3?"medium":"bad";Check(f.Text("SELECT data FROM statistics_events WHERE event_type='game_completed'").Contains(expected),"ending rule unchanged");
   var batch=f.Stats.Reserve(1);Check(batch.Count<=5&&batch.Count>0,"small fixed batch");

   var first=batch[0];Reject(()=>f.Stats.Confirm(first.Event.EventId,first.LeaseId,new string('0',64)),"STATISTICS_RECEIPT_MISMATCH");
   f.Stats.Fail(first.Event.EventId,first.LeaseId,2,"quota_exceeded");Check(f.Stats.Reserve(100000).Count==0,"quota pauses FIFO without dropping pending");
   var reopen=new StatisticsTelemetry(f.Db);reopen.ResumeQuota(100001);var retry=reopen.Reserve(100002);Check(retry.Count>0&&retry[0].Event==first.Event&&retry[0].PayloadHash==first.PayloadHash,"reopen and retry keep event and hash");

 static async Task InterruptedAndAnimal(){
  using var f=new F(false);await using var a=GamePersistenceAdapter.CreateSyntheticStatisticsAdapter(f.Db);await a.OpenAsync("owned-outbox-fixture");await a.CreateLocalProfileAsync(new(Id(),f.Owner,"Synthetic"));await a.CreateNewGameAsync(new(Id(),f.Owner,f.Game));

  long clock=0;var first=new LocalGameplayObserver(a,()=>clock);await first.InitializeAsync(f.Owner);first.AnimalStarted(f.Game,"ocean","animal1");first.Observe(true,false,false,true);clock=30000;first.Observe(true,false,false,true);await first.DrainAsync();
  first.Died();await first.DrainAsync();first.RequestRestart(true);await first.DrainAsync();first.AnimalStarted(f.Game,"ocean","animal2");first.Observe(true,false,false,true);clock+=1000;first.Leave();await first.DrainAsync();
  Check(f.N("SELECT sum(duration_ms) FROM statistics_events WHERE event_type='playtime_interval'")==31000,"animal death/retry does not double-count prior segment");
  // Simulate process loss by abandoning the first observer, not fabricating a terminal event or elapsed UTC duration.
  await a.DisposeAsync();await using var reopened=GamePersistenceAdapter.CreateSyntheticStatisticsAdapter(f.Db);await reopened.OpenAsync("owned-outbox-fixture");
  var second=new LocalGameplayObserver(reopened,()=>clock);await second.InitializeAsync(f.Owner);await second.DrainAsync();
  Check(f.N("SELECT count(*) FROM gameplay_sessions WHERE status='interrupted'")==1,"unexpected close marks old session interrupted");
  Check(f.N("SELECT sum(duration_ms) FROM statistics_events WHERE event_type='playtime_interval'")==31000,"reopen never invents time between processes");
  await second.CloseAsync();Check(f.N("SELECT count(*) FROM statistics_events WHERE event_type='session_started'")==2,"new statistical session on reopen");
  var events=new GameplayEventService(f.Db);var sess=Id();events.RecordEvent(new(Id(),f.Owner,sess,GameplayEventKind.SESSION_STARTED,DateTimeOffset.UtcNow,0));
  Reject(()=>events.RecordInterval(new(Id(),f.Owner,sess,-1,0,GameplayTimeState.Menu)),"INVALID_EVENT");
 }
